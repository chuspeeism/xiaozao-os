优雅不是少写一行，而是让下一个看到它的人，会心一笑。这里是一个 Node.js 的内存 LRU 缓存。

lru-cache.js  ·  Node.js 
'use strict';

const { EventEmitter } = require('events');

/**
 * 一个简单的内存 LRU 缓存。
 * 利用 Map 的插入顺序来追踪“最近使用”。
 */
class LruCache extends EventEmitter {
  constructor(capacity = 100) {
    super();
    this.capacity = capacity;
    this.store = new Map();
  }

  get size() {
    return this.store.size;
  }

  get(key) {
    if (!this.store.has(key)) {
      this.emit('miss', key);
      return undefined;
    }
    // 命中后把它移到“最新”的位置。
    const value = this.store.get(key);
    this.store.delete(key);
    this.store.set(key, value);
    this.emit('hit', key);
    return value;
  }

  set(key, value) {
    if (this.store.has(key)) {
      this.store.delete(key);
    } else if (this.store.size >= this.capacity) {
      // Map 的第一个 key 就是最久没用的。
      const oldest = this.store.keys().next().value;
      this.store.delete(oldest);
      this.emit('evict', oldest);
    }
    this.store.set(key, value);
    return this;
  }

  has(key) {
    return this.store.has(key);
  }

  clear() {
    this.store.clear();
    this.emit('clear');
  }
}

module.exports = LruCache;

①先看一下这个方程

同学们，上课我们谈了黎曼几何。今天主角来了——爱因斯坦场方程。他说：物质告诉时空如何弯曲，时空告诉物质如何运动。

—— 爱因斯坦场方程
$$ R_{\mu\nu} - \tfrac{1}{2} g_{\mu\nu} R + \Lambda g_{\mu\nu} \;=\; \frac{8\pi G}{c^{4}} \, T_{\mu\nu} $$

记住：左边是几何，右边是物质。等号把宇宙缝在了一起。

②那 R 是什么？

拆开看：里奇张量 Rμν 是由克里斯托特菲符号 Γ 逐层求导得来的。写出来如下——看起来复杂，但每一项都在描述同一件事：坐标变换带来的“偏差”。

$$ R_{\mu\nu} \;=\; \partial_{\alpha} \Gamma^{\alpha}_{\mu\nu} - \partial_{\nu} \Gamma^{\alpha}_{\mu\alpha} + \Gamma^{\alpha}_{\alpha\beta}\Gamma^{\beta}_{\mu\nu} - \Gamma^{\alpha}_{\nu\beta}\Gamma^{\beta}_{\mu\alpha} $$

③从作用量重新推一遇

这个方程不是猜出来的。如果我们写下 Einstein–Hilbert 作用量，并对 gμν 取变分，它会自动掉出来。

—— Einstein–Hilbert 作用量
$$ S \;=\; \frac{c^{4}}{16\pi G} \!\int \! \left( R - 2\Lambda \right) \sqrt{-g}\, \mathrm{d}^{4}x \;+\; \int \! \mathcal{L}_{\text{M}} \sqrt{-g}\, \mathrm{d}^{4}x $$

课后思考：为什么一定要乘 √(-g) ？

④一个具体的解

把场方程应用到“球对称、静态、真空”这样最简单的场景上，可以解出史瓦西希度规——黑洞就住在它里。

—— Schwarzschild 度规
$$ \mathrm{d}s^{2} = -\!\left(1 - \tfrac{2GM}{c^{2}r}\right)\! c^{2}\mathrm{d}t^{2} + \left(1 - \tfrac{2GM}{c^{2}r}\right)^{\!-1}\!\mathrm{d}r^{2} + r^{2}\!\mathrm{d}\Omega^{2} $$
